cmake_minimum_required(VERSION 3.31)
project(LexiDictionaryReader LANGUAGES C CXX)
set(MKD_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(MKD_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MKD_SOURCE}" mkd EXCLUDE_FROM_ALL)
add_executable(lexi-dictionary-reader main.cpp)
target_link_libraries(lexi-dictionary-reader PRIVATE MKD::mkd)
